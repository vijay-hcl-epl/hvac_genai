Example C Unit Test Stub for UART Command Handler
--------------------------------------------------------------------------------
#include "unity.h"
#include "uart_command_handler.h"
#include "mock_flap_control_logic.h"

void setUp(void) { UART_CommandHandler_Init(); }
void tearDown(void) {}

void test_UART_valid_command(void) {
    int cmd;
    UART_CommandHandler_ReceiveChar('4');
    UART_CommandHandler_ReceiveChar('\n');
    TEST_ASSERT_TRUE(UART_CommandHandler_GetValidCommand(&cmd));
    TEST_ASSERT_EQUAL_INT(4, cmd);
}

void test_UART_invalid_command(void) {
    int cmd;
    UART_CommandHandler_ReceiveChar('8');
    UART_CommandHandler_ReceiveChar('\n');
    TEST_ASSERT_FALSE(UART_CommandHandler_GetValidCommand(&cmd));
}
