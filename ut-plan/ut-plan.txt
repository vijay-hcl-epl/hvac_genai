SECTION 4: EXAMPLE UNIT TEST STUB (C, Using Mocks)

#include <unity.h>
#include "command_interface.h"
#include "mock_uart_hw.h"

void setUp(void) { command_interface_init(); }
void tearDown(void) { }

void test_command_interface_valid_command(void) {
    uint8_t frame[8] = {CMD_TYPE_POSITION, 0x20, 0x00, 0,0,0,0,0};
    for (int i=0;i<8;i++) mock_uart_hw_receive(frame[i]);
    command_t cmd = command_interface_get_valid_command();
    TEST_ASSERT_TRUE(cmd.valid_command_flag);
    TEST_ASSERT_EQUAL(CMD_TYPE_POSITION, cmd.type);
}
