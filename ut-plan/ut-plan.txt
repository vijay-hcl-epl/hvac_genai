4. Isolation Rules
--------------------
- All hardware dependencies (UART, LIN, ADC1, GPIO, motor driver HW) are to be replaced by mocks/stubs—NO real hardware accessed.
- Unit tests to run under deterministic, repeatable harness.
- No integration tests in this plan—unit scope only.
- No interaction with boot code, main loop scheduling.

// command_handler_test.c
#include "unity.h"
#include "command_handler.h"
#include "mock_config_manager.h"
#include "mock_status_indication.h"

void test_Command_Receive_valid_range(void) {
    Config_IsValidCommand_ExpectAndReturn(0, 1);
    Command_Receive(0);
    TEST_ASSERT_EQUAL_UINT8(1, Command_Validate());
    TEST_ASSERT_EQUAL_UINT8(0, Command_GetError());
}

void test_Command_Receive_invalid_range(void) {
    Command_Receive(0xFF);
    TEST_ASSERT_EQUAL_UINT8(0, Command_Validate());
    TEST_ASSERT_EQUAL_UINT8(1, Command_GetError());
}
