// Example stub for Command Handler Unit
#include "unity.h"
#include "command_handler.h"
void test_CommandHandler_ValidCommand(void) {
    CommandHandler_Init();
    CommandHandler_UartRxHandler('4');
    int cmd = CommandHandler_GetValidCommand();
    TEST_ASSERT_EQUAL(4, cmd);
}
void test_CommandHandler_InvalidCommand(void) {
    CommandHandler_Init();
    CommandHandler_UartRxHandler('x');
    int cmd = CommandHandler_GetValidCommand();
    TEST_ASSERT_EQUAL(CMD_INVALID, cmd);
}
