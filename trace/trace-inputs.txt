#include "flap_controller.h"

// Internal static instance
static FlapController_t g_flap_controller = {0, 0, FLAP_IDLE};

// Simulate reading command and performing validation (SWE-REQ-001, SWE-REQ-002)
void FlapController_ReceiveAndValidateCommand(uint8_t requested_position) {
    // Only allow commands within 0-100
    if(requested_position <= 100) {
        g_flap_controller.requested_position = requested_position;
        g_flap_controller.state = FLAP_IDLE;
    } else {
        g_flap_controller.state = FLAP_INVALID_COMMAND;
    }
}

// Simulate reading of flap potentiometer (SWE-REQ-003)
uint8_t FlapController_ReadCurrentPosition(void) {
    // For integration, this would read ADC or similar HW
    return g_flap_controller.current_position;
}

// Drives the flap motor towards requested position (SWE-REQ-004, SWE-REQ-005, SWE-REQ-006)
void FlapController_UpdateMotor(uint8_t requested_position) {
    if(g_flap_controller.state == FLAP_INVALID_COMMAND) {
        return; // Do not operate on invalid commands
    }
    if(g_flap_controller.current_position < requested_position) {
        g_flap_controller.current_position++;
        g_flap_controller.state = FLAP_MOVING;
    } else if(g_flap_controller.current_position > requested_position) {
        g_flap_controller.current_position--;
        g_flap_controller.state = FLAP_MOVING;
    } else {
        g_flap_controller.state = FLAP_TARGET_REACHED;
        // SWE-REQ-006: Stop the flap motor
    }
}
