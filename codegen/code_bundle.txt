/**
 * @file command_parser.h
 * @brief Command Parser – UART byte parsing and validation for flap position commands.
 *
 * Software Unit : Command Parser
 * Traces        : SWE-REQ-001, SWE-REQ-002, SWE-REQ-003, SWE-REQ-015, SWE-REQ-022
 * Platform      : STM32F407G-DISC1, STM32 HAL
 */

#ifndef COMMAND_PARSER_H
#define COMMAND_PARSER_H

#include <stdint.h>

/** Number of supported flap positions (SWE-REQ-027, SWE-REQ-030) */
#define CMD_NUM_POSITIONS  ((uint8_t)5U)

/** Position command byte values sent from PC over UART (SWE-REQ-001) */
#define CMD_POS_FULLY_CLOSED  ((uint8_t)0x00U)
#define CMD_POS_25_PERCENT    ((uint8_t)0x01U)
#define CMD_POS_50_PERCENT    ((uint8_t)0x02U)
#define CMD_POS_75_PERCENT    ((uint8_t)0x03U)
#define CMD_POS_FULLY_OPEN    ((uint8_t)0x04U)

/** Command parser states (LLD state machine: INIT, WAIT_RX, VALID, INVALID) */
typedef enum
{
    CMD_STATE_INIT    = 0U,
    CMD_STATE_WAIT_RX,
    CMD_STATE_VALID,
    CMD_STATE_INVALID
} CmdParserState_t;

/** Command result structure (SWE-REQ-009, SWE-REQ-036, SWE-REQ-038) */
typedef struct
{
    uint8_t          position;   /**< Last valid position command byte */
    uint8_t          valid;      /**< 1 = valid command available, 0 = none */
    CmdParserState_t state;      /**< Current parser state */
} CmdParserData_t;

/**
 * @brief Initialise the command parser module.
 * Traces: SWE-REQ-020 (called by system_init)
 */
void CommandParser_Init(void);

/**
 * @brief Process one pending UART receive byte (non-blocking).
 *        Must be called cyclically from the main loop.
 * Traces: SWE-REQ-001, SWE-REQ-002, SWE-REQ-003, SWE-REQ-022
 */
void CommandParser_Process(void);

/**
 * @brief Retrieve the latest validated command.
 * @param[out] pPosition  Pointer to receive the position value.
 * @return 1 if a valid unread command is available, 0 otherwise.
 * Traces: SWE-REQ-001, SWE-REQ-003, SWE-REQ-015
 */
uint8_t CommandParser_GetLatestCommand(uint8_t *pPosition);

#endif /* COMMAND_PARSER_H */

/**
 * @file command_parser.c
 * @brief Command Parser implementation.
 *
 * Software Unit : Command Parser
 * Traces        : SWE-REQ-001, SWE-REQ-002, SWE-REQ-003, SWE-REQ-015, SWE-REQ-022
 * Platform      : STM32F407G-DISC1, STM32 HAL (UART)
 */

#include "command_parser.h"
#include "stm32f4xx_hal.h"

/* ---------- Platform binding ------------------------------------------- */
/* UART handle – defined externally by STM32 HAL / CubeMX generated code   */
extern UART_HandleTypeDef huart2;  /* USART2 on STM32F407-DISC1 (ST-Link VCP) */

/* ---------- Static data (SWE-REQ-009, SWE-REQ-036, SWE-REQ-038) ------- */
static CmdParserData_t cmdData;
static uint8_t         rxByte;     /* Single-byte receive buffer */

/* Allowed position values table (SWE-REQ-027, SWE-REQ-030) */
static const uint8_t allowedPositions[CMD_NUM_POSITIONS] =
{
    CMD_POS_FULLY_CLOSED,
    CMD_POS_25_PERCENT,
    CMD_POS_50_PERCENT,
    CMD_POS_75_PERCENT,
    CMD_POS_FULLY_OPEN
};

/* ---------- Private helpers -------------------------------------------- */

/**
 * @brief Check if a byte is a valid position command.
 * Traces: SWE-REQ-002
 */
static uint8_t IsValidPosition(uint8_t byte)
{
    uint8_t i;
    for (i = 0U; i < CMD_NUM_POSITIONS; i++)
    {
        if (byte == allowedPositions[i])
        {
            return 1U;
        }
    }
    return 0U;
}

/* ---------- Public API ------------------------------------------------- */

void CommandParser_Init(void)
{
    cmdData.position = 0U;
    cmdData.valid    = 0U;
    cmdData.state    = CMD_STATE_WAIT_RX;
    rxByte           = 0U;
}

void CommandParser_Process(void)
{
    HAL_StatusTypeDef status;

    /* Non-blocking single-byte receive (SWE-REQ-022) */
    status = HAL_UART_Receive(&huart2, &rxByte, 1U, 0U);

    if (status == HAL_OK)
    {
        if (IsValidPosition(rxByte) != 0U)
        {
            /* Latest-wins: overwrite previous command (SWE-REQ-003) */
            cmdData.position = rxByte;
            cmdData.valid    = 1U;
            cmdData.state    = CMD_STATE_VALID;
        }
        else
        {
            /* Invalid / unsupported – ignore, no acknowledge (SWE-REQ-002) */
            cmdData.state = CMD_STATE_INVALID;
        }
    }
    /* If HAL_TIMEOUT or HAL_BUSY: no byte available – remain in WAIT_RX */

    if (cmdData.state == CMD_STATE_INVALID)
    {
        /* Flush: return to wait state (LLD transition INVALID->WAIT_RX) */
        cmdData.state = CMD_STATE_WAIT_RX;
    }
}

uint8_t CommandParser_GetLatestCommand(uint8_t *pPosition)
{
    if ((pPosition != ((void *)0)) && (cmdData.valid == 1U))
    {
        *pPosition    = cmdData.position;
        cmdData.valid = 0U;   /* Consume the command */
        return 1U;
    }
    return 0U;
}

/**
 * @file feedback_processor.c
 * @brief Feedback Processor implementation.
 *
 * Software Unit : Feedback Processor
 * Traces        : SWE-REQ-010, SWE-REQ-011, SWE-REQ-017, SWE-REQ-027,
 *                 SWE-REQ-028, SWE-REQ-030
 * Platform      : STM32F407G-DISC1, STM32 HAL ADC, 12-bit
 */

#include "feedback_processor.h"
#include "stm32f4xx_hal.h"

/* ---------- Platform binding ------------------------------------------- */
extern ADC_HandleTypeDef hadc1;  /* ADC1 – potentiometer feedback channel */

/* ---------- Static position mapping table (SWE-REQ-027, SWE-REQ-030) --- */
/**
 * Each entry: { lowThreshold, highThreshold }
 * Zones divide the 12-bit ADC range (0..4095) into 5 equal bands.
 * SWE-REQ-028: No self-learning; table is fixed at compile time.
 */
typedef struct
{
    uint16_t low;
    uint16_t high;
} AdcZone_t;

static const AdcZone_t positionMap[FB_NUM_POSITIONS] =
{
    {    0U,   819U },   /* Position 0 – Fully closed  */
    {  820U,  1638U },   /* Position 1 – 25 %          */
    { 1639U,  2457U },   /* Position 2 – 50 %          */
    { 2458U,  3276U },   /* Position 3 – 75 %          */
    { 3277U,  4095U }    /* Position 4 – Fully open    */
};

/* ---------- Static data (SWE-REQ-036, SWE-REQ-038) --------------------- */
static FeedbackData_t fbData;

/* ---------- Public API ------------------------------------------------- */

void FeedbackProcessor_Init(void)
{
    fbData.rawAdc   = 0U;
    fbData.position = 0U;
    fbData.valid    = 0U;
    fbData.state    = FB_STATE_INIT;
}

void FeedbackProcessor_Update(void)
{
    HAL_StatusTypeDef halStatus;
    uint16_t adcValue;
    uint8_t  i;

    /* Start conversion and poll (blocking, single-channel) – SWE-REQ-017 */
    HAL_ADC_Start(&hadc1);
    halStatus = HAL_ADC_PollForConversion(&hadc1, 10U);  /* 10 ms timeout */

    if (halStatus == HAL_OK)
    {
        adcValue = (uint16_t)HAL_ADC_GetValue(&hadc1);

        /* Bounding / validation – SWE-REQ-011 */
        if (adcValue > FB_ADC_MAX)
        {
            fbData.valid = 0U;
            fbData.state = FB_STATE_INVALID;
        }
        else
        {
            fbData.rawAdc = adcValue;

            /* Table lookup – SWE-REQ-010, SWE-REQ-027 */
            fbData.valid = 0U;  /* Assume invalid until matched */
            for (i = 0U; i < FB_NUM_POSITIONS; i++)
            {
                if ((adcValue >= positionMap[i].low) &&
                    (adcValue <= positionMap[i].high))
                {
                    fbData.position = i;
                    fbData.valid    = 1U;
                    fbData.state    = FB_STATE_READY;
                    break;
                }
            }
            if (fbData.valid == 0U)
            {
                /* Value within ADC range but not in any zone – mark invalid */
                fbData.state = FB_STATE_INVALID;
            }
        }
    }
    else
    {
        /* ADC conversion failed – mark invalid (SWE-REQ-031) */
        fbData.valid = 0U;
        fbData.state = FB_STATE_INVALID;
    }

    HAL_ADC_Stop(&hadc1);
}

uint8_t FeedbackProcessor_GetPosition(uint8_t *pPosition)
{
    if ((pPosition != ((void *)0)) && (fbData.valid == 1U))
    {
        *pPosition = fbData.position;
        return 1U;
    }
    return 0U;
}

/**
 * @file flap_control_logic.c
 * @brief Flap Control Logic implementation.
 *
 * Software Unit : Flap Control Logic
 * Traces        : SWE-REQ-004, SWE-REQ-005, SWE-REQ-006, SWE-REQ-023,
 *                 SWE-REQ-024, SWE-REQ-031
 * Platform      : STM32F407G-DISC1
 */

#include "flap_control_logic.h"
#include "command_parser.h"
#include "feedback_processor.h"
#include "motor_driver.h"
#include "led_status_handler.h"

/* ---------- Static data (SWE-REQ-009, SWE-REQ-036) --------------------- */
static FlapControlData_t flapData;

/* ---------- Public API ------------------------------------------------- */

void FlapControl_Init(void)
{
    flapData.currentPos = 0U;
    flapData.targetPos  = 0U;
    flapData.inMotion   = 0U;
    flapData.state      = FLAP_STATE_IDLE;
}

void FlapControl_Run(void)
{
    uint8_t newCmd;
    uint8_t cmdAvailable;
    uint8_t fbPos;
    uint8_t fbValid;

    /* ---- 1. Acquire latest feedback (SWE-REQ-023) ---- */
    FeedbackProcessor_Update();
    fbValid = FeedbackProcessor_GetPosition(&fbPos);

    /* ---- 2. Check for feedback fault (SWE-REQ-031, SWE-REQ-025) ---- */
    if (fbValid == 0U)
    {
        /* Fault path: stop motor immediately, set fault state */
        MotorDriver_Stop();
        flapData.inMotion = 0U;
        flapData.state    = FLAP_STATE_FAULT;
        LedStatus_IndicateError();
        return;
    }

    /* Update known current position */
    flapData.currentPos = fbPos;

    /* ---- 3. Process new command if available (SWE-REQ-001) ---- */
    cmdAvailable = CommandParser_GetLatestCommand(&newCmd);
    if (cmdAvailable == 1U)
    {
        flapData.targetPos = newCmd;
    }

    /* ---- 4. State machine (LLD: IDLE, MOVING, TARGET_REACHED, FAULT) ---- */
    switch (flapData.state)
    {
        case FLAP_STATE_IDLE:
            /* Fall-through to evaluate if movement is needed */
        case FLAP_STATE_TARGET_REACHED:
            if (flapData.targetPos != flapData.currentPos)  /* SWE-REQ-006 */
            {
                /* Determine direction (SWE-REQ-004) */
                if (flapData.targetPos > flapData.currentPos)
                {
                    MotorDriver_Drive(MOTOR_DIR_FORWARD, 1U);
                }
                else
                {
                    MotorDriver_Drive(MOTOR_DIR_REVERSE, 1U);
                }
                flapData.inMotion = 1U;
                flapData.state    = FLAP_STATE_MOVING;
            }
            else
            {
                /* Already at target – ensure correct LED indication (SWE-REQ-013) */
                LedStatus_SetPosition(flapData.currentPos);
            }
            break;

        case FLAP_STATE_MOVING:
            /* Check if target reached (SWE-REQ-005, SWE-REQ-024) */
            if (flapData.currentPos == flapData.targetPos)
            {
                MotorDriver_Stop();  /* SWE-REQ-008 */
                flapData.inMotion = 0U;
                flapData.state    = FLAP_STATE_TARGET_REACHED;
                LedStatus_SetPosition(flapData.currentPos);  /* SWE-REQ-013 */
            }
            else
            {
                /* Still moving – re-evaluate direction in case of new command */
                if (flapData.targetPos > flapData.currentPos)
                {
                    MotorDriver_Drive(MOTOR_DIR_FORWARD, 1U);
                }
                else
                {
                    MotorDriver_Drive(MOTOR_DIR_REVERSE, 1U);
                }
            }
            break;

        case FLAP_STATE_FAULT:
            /* Recovery: feedback is valid again (checked above), return to idle */
            MotorDriver_Stop();
            flapData.inMotion = 0U;
            flapData.state    = FLAP_STATE_IDLE;
            LedStatus_SetPosition(flapData.currentPos);
            break;

        default:
            /* Defensive: stop motor, reset to idle */
            MotorDriver_Stop();
            flapData.inMotion = 0U;
            flapData.state    = FLAP_STATE_IDLE;
            break;
    }
}

FlapControlState_t FlapControl_GetState(void)
{
    return flapData.state;
}

/**
 * @file led_status_handler.c
 * @brief LED Status Handler implementation.
 *
 * Software Unit : LED Status Handler
 * Traces        : SWE-REQ-012, SWE-REQ-013, SWE-REQ-014, SWE-REQ-018,
 *                 SWE-REQ-021, SWE-REQ-039
 * Platform      : STM32F407G-DISC1, GPIO-based LEDs
 */

#include "led_status_handler.h"
#include "stm32f4xx_hal.h"

/* ---------- Platform GPIO mapping (env-setup: GPIO for LED control) ---- */
/*
 * STM32F407G-DISC1 on-board LEDs:
 *   PD12 – Green   (used as Power / Initialization LED)
 *   PD13 – Orange  (position 0 indicator – Fully closed)
 *   PD14 – Red     (position 1 indicator – 25%)
 *   PD15 – Blue    (position 2 indicator – 50%)
 * External LEDs (directly driven GPIOs):
 *   PE0  – Green   (position 3 indicator – 75%)
 *   PE1  – Green   (position 4 indicator – Fully open)
 *
 * All active-high.
 */

/** Power LED */
#define LED_POWER_PORT  GPIOD
#define LED_POWER_PIN   GPIO_PIN_12

/** Position indicator LED table – index matches logical position 0..4 */
typedef struct
{
    GPIO_TypeDef *port;
    uint16_t      pin;
} LedGpioMap_t;

static const LedGpioMap_t posLedMap[LED_NUM_POSITION_LEDS] =
{
    { GPIOD, GPIO_PIN_13 },   /* Position 0 – Fully closed */
    { GPIOD, GPIO_PIN_14 },   /* Position 1 – 25 %         */
    { GPIOD, GPIO_PIN_15 },   /* Position 2 – 50 %         */
    { GPIOE, GPIO_PIN_0  },   /* Position 3 – 75 %         */
    { GPIOE, GPIO_PIN_1  }    /* Position 4 – Fully open   */
};

/* ---------- Static data (SWE-REQ-036, SWE-REQ-038) --------------------- */
static uint8_t currentLedPos;  /* Last indicated position */

/* ---------- Private helpers -------------------------------------------- */

/**
 * @brief Turn all position LEDs OFF.
 * Traces: SWE-REQ-013 (one-hot enforcement)
 */
static void AllPositionLedsOff(void)
{
    uint8_t i;
    for (i = 0U; i < LED_NUM_POSITION_LEDS; i++)
    {
        HAL_GPIO_WritePin(posLedMap[i].port, posLedMap[i].pin, GPIO_PIN_RESET);
    }
}

/* ---------- Public API ------------------------------------------------- */

void LedStatus_Init(void)
{
    currentLedPos = 0xFFU;  /* Force first update */
    AllPositionLedsOff();
    /* Power LED ON at startup (SWE-REQ-012) */
    LedStatus_SetPowerLed();
}

void LedStatus_SetPosition(uint8_t position)
{
    /* Only alter LEDs on actual change (SWE-REQ-014) */
    if ((position < LED_NUM_POSITION_LEDS) && (position != currentLedPos))
    {
        AllPositionLedsOff();  /* One-hot: SWE-REQ-013 */
        HAL_GPIO_WritePin(posLedMap[position].port,
                          posLedMap[position].pin,
                          GPIO_PIN_SET);
        currentLedPos = position;
    }
}

void LedStatus_SetPowerLed(void)
{
    HAL_GPIO_WritePin(LED_POWER_PORT, LED_POWER_PIN, GPIO_PIN_SET);
}

void LedStatus_IndicateError(void)
{
    /*
     * Error indication: all position LEDs OFF, power LED stays ON.
     * SWE-REQ-032: safe idle – no movement, status LEDs unchanged beyond
     * clearing position indicator.
     */
    AllPositionLedsOff();
    currentLedPos = 0xFFU;
}

/**
 * @file main.c
 * @brief Application entry point – init + super-loop task dispatcher.
 *
 * Traces: SWE-REQ-020 (startup), SWE-REQ-022 (constant monitoring),
 *         SWE-REQ-033 (responsiveness), SWE-REQ-034 (no assumed execution model)
 * Platform: STM32F407G-DISC1, STM32 HAL, bare-metal super-loop
 *
 * No RTOS constructs are used (SWE-REQ-026).
 * No dynamic memory allocation (SWE-REQ-036).
 */

#include "stm32f4xx_hal.h"
#include "system_init.h"
#include "command_parser.h"
#include "flap_control_logic.h"

/**
 * @brief Application entry point.
 *
 * Performs system initialisation then enters an infinite super-loop
 * that continuously dispatches the command parser and flap control logic.
 *
 * @return int  (never reached in bare-metal operation)
 */
int main(void)
{
    /* ---- System initialisation (SWE-REQ-020) ---- */
    SystemInit_Run();

    /* ---- Main task dispatcher – super-loop (SWE-REQ-022, SWE-REQ-033) ---- */
    while (1)
    {
        /* 1. Process any pending UART command (SWE-REQ-022) */
        CommandParser_Process();

        /* 2. Run flap control state machine (SWE-REQ-004, SWE-REQ-023) */
        FlapControl_Run();
    }

    /* Unreachable – required by MISRA for int main return */
    return 0;
}

/**
 * @file motor_driver.c
 * @brief Motor Driver implementation – L298N control via STM32 GPIO/PWM.
 *
 * Software Unit : Motor Driver
 * Traces        : SWE-REQ-007, SWE-REQ-008, SWE-REQ-016, SWE-REQ-025,
 *                 SWE-REQ-036, SWE-REQ-038
 * Platform      : STM32F407G-DISC1, L298N H-Bridge, Timer-based PWM
 */

#include "motor_driver.h"
#include "stm32f4xx_hal.h"

/* ---------- Platform GPIO / Timer mapping (env-setup) ------------------ */
/*
 * L298N connections on STM32F407G-DISC1:
 *   IN1 – PB0  (direction A)    GPIO output
 *   IN2 – PB1  (direction B)    GPIO output
 *   ENA – PA0  (PWM enable)     TIM2 Channel 1
 *
 * Motor Type: 12 V DC geared motor (env-setup)
 */
#define MOTOR_IN1_PORT   GPIOB
#define MOTOR_IN1_PIN    GPIO_PIN_0
#define MOTOR_IN2_PORT   GPIOB
#define MOTOR_IN2_PIN    GPIO_PIN_1

extern TIM_HandleTypeDef htim2;  /* TIM2 – PWM for motor enable */
#define MOTOR_PWM_CHANNEL  TIM_CHANNEL_1

/* ---------- Static data (SWE-REQ-036, SWE-REQ-038) --------------------- */
static MotorDriverData_t motorData;

/* ---------- Private helpers -------------------------------------------- */

/**
 * @brief Set direction GPIO pins for L298N.
 * Traces: SWE-REQ-007, SWE-REQ-016
 */
static void SetDirectionPins(MotorDirection_t dir)
{
    switch (dir)
    {
        case MOTOR_DIR_FORWARD:
            HAL_GPIO_WritePin(MOTOR_IN1_PORT, MOTOR_IN1_PIN, GPIO_PIN_SET);
            HAL_GPIO_WritePin(MOTOR_IN2_PORT, MOTOR_IN2_PIN, GPIO_PIN_RESET);
            break;

        case MOTOR_DIR_REVERSE:
            HAL_GPIO_WritePin(MOTOR_IN1_PORT, MOTOR_IN1_PIN, GPIO_PIN_RESET);
            HAL_GPIO_WritePin(MOTOR_IN2_PORT, MOTOR_IN2_PIN, GPIO_PIN_SET);
            break;

        case MOTOR_DIR_STOP: /* fall-through */
        default:
            HAL_GPIO_WritePin(MOTOR_IN1_PORT, MOTOR_IN1_PIN, GPIO_PIN_RESET);
            HAL_GPIO_WritePin(MOTOR_IN2_PORT, MOTOR_IN2_PIN, GPIO_PIN_RESET);
            break;
    }
}

/* ---------- Public API ------------------------------------------------- */

void MotorDriver_Init(void)
{
    motorData.direction = MOTOR_DIR_STOP;
    motorData.enable    = 0U;
    motorData.pwmDuty   = 0U;
    motorData.status    = MOTOR_STATUS_IDLE;

    /* Ensure motor is OFF at init (SWE-REQ-020) */
    SetDirectionPins(MOTOR_DIR_STOP);
    __HAL_TIM_SET_COMPARE(&htim2, MOTOR_PWM_CHANNEL, 0U);
    HAL_TIM_PWM_Start(&htim2, MOTOR_PWM_CHANNEL);
}

void MotorDriver_Drive(MotorDirection_t dir, uint8_t enable)
{
    if ((dir == MOTOR_DIR_FORWARD) || (dir == MOTOR_DIR_REVERSE))
    {
        motorData.direction = dir;
        motorData.enable    = enable;

        SetDirectionPins(dir);

        if (enable != 0U)
        {
            motorData.pwmDuty = MOTOR_DEFAULT_PWM_DUTY;
            __HAL_TIM_SET_COMPARE(&htim2, MOTOR_PWM_CHANNEL,
                                  (uint32_t)motorData.pwmDuty);
            motorData.status = MOTOR_STATUS_RUNNING;
        }
        else
        {
            __HAL_TIM_SET_COMPARE(&htim2, MOTOR_PWM_CHANNEL, 0U);
            motorData.status = MOTOR_STATUS_IDLE;
        }
    }
    else
    {
        /* Invalid direction – stop (SWE-REQ-008) */
        MotorDriver_Stop();
    }
}

void MotorDriver_Stop(void)
{
    motorData.direction = MOTOR_DIR_STOP;
    motorData.enable    = 0U;
    motorData.pwmDuty   = 0U;
    motorData.status    = MOTOR_STATUS_IDLE;

    SetDirectionPins(MOTOR_DIR_STOP);
    __HAL_TIM_SET_COMPARE(&htim2, MOTOR_PWM_CHANNEL, 0U);
}

MotorStatus_t MotorDriver_GetStatus(void)
{
    return motorData.status;
}

/**
 * @file system_init.c
 * @brief System Startup / Initialisation implementation.
 *
 * Software Unit : System Startup/Initialization
 * Traces        : SWE-REQ-020
 * Platform      : STM32F407G-DISC1, STM32 HAL
 */

#include "system_init.h"
#include "stm32f4xx_hal.h"
#include "command_parser.h"
#include "feedback_processor.h"
#include "flap_control_logic.h"
#include "motor_driver.h"
#include "led_status_handler.h"

/* ---------- Extern HAL handles (generated by STM32CubeMX) -------------- */
extern UART_HandleTypeDef  huart2;
extern ADC_HandleTypeDef   hadc1;
extern TIM_HandleTypeDef   htim2;

/* ---------- Forward declarations for CubeMX-style peripheral init ------ */
/*
 * In a full STM32CubeMX project these functions are auto-generated.
 * Here we declare them extern so the linker resolves them from the
 * CubeMX-generated code.  The bodies are NOT part of the application
 * software units (they are platform/BSP).
 */
extern void SystemClock_Config(void);
extern void MX_GPIO_Init(void);
extern void MX_USART2_UART_Init(void);
extern void MX_ADC1_Init(void);
extern void MX_TIM2_Init(void);

/* ---------- Static data (SWE-REQ-036, SWE-REQ-038) --------------------- */
static uint8_t initDone;  /* 0 = not done, 1 = complete */

/* ---------- Public API ------------------------------------------------- */

void SystemInit_Run(void)
{
    if (initDone != 0U)
    {
        return;  /* Guard: single-pass only (LLD: not re-entrant) */
    }

    /* ---- HAL & clock initialisation ---- */
    HAL_Init();
    SystemClock_Config();

    /* ---- Peripheral initialisation (CubeMX-generated) ---- */
    MX_GPIO_Init();
    MX_USART2_UART_Init();
    MX_ADC1_Init();
    MX_TIM2_Init();

    /* ---- Application unit initialisation (SWE-REQ-020) ---- */
    CommandParser_Init();
    FeedbackProcessor_Init();
    FlapControl_Init();
    MotorDriver_Init();      /* Sets motor OFF / safe state */
    LedStatus_Init();        /* Power LED on, position LEDs off */

    /* ---- Read initial feedback & indicate on LEDs (SWE-REQ-021) ---- */
    {
        uint8_t initPos;
        FeedbackProcessor_Update();
        if (FeedbackProcessor_GetPosition(&initPos) == 1U)
        {
            LedStatus_SetPosition(initPos);
        }
        else
        {
            LedStatus_IndicateError();
        }
    }

    initDone = 1U;
}
