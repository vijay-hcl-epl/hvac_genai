###### FILE: code/src/command_handler.c ######
#include "command_handler.h"
#include "status_indication.h"
#include "config_manager.h"
#include <stdint.h>

static uint8_t command_received = 0u;
static uint8_t command_validated = 0u;
static uint8_t command_last = 0u;
static uint8_t error_code = 0u;

void Command_Receive(uint8_t data) {
    /* Only 0..5 are valid, per mapping table loaded from Flash */
    const uint8_t min_cmd = 0u;
    const uint8_t max_cmd = 5u;
    if (data >= min_cmd && data <= max_cmd) {
        /* Lookup in command mapping table from Flash */
        if (Config_IsValidCommand(data)) {
            command_received = data;
            command_validated = 1u;
            error_code = 0u;
        } else {
            error_code = 2u; /* invalid mapping */
            Status_SetError(error_code);
        }
    } else {
        error_code = 1u; /* out of range */
        Status_SetError(error_code);
        return; /* ignore, per SWE-REQ-002 */
    }
}

uint8_t Command_Validate(void) {
    return command_validated;
}

uint8_t Command_GetError(void) {
    return error_code;
}

#include "position_control.h"
#include "feedback_acquisition.h"
#include "motor_driver.h"
#include "status_indication.h"
#include <stdint.h>

static uint8_t target_position = 0u;
static uint8_t current_position = 0u;
static uint8_t last_position = 0u;
static uint8_t movement_pending = 0u;
static uint8_t holding = 1u;

void Control_Request(uint8_t new_pos) {
    current_position = Feedback_Get();
    if (new_pos == current_position) {
        holding = 1u;
        movement_pending = 0u;
        return;
    }
    target_position = new_pos;
    movement_pending = 1u;
    holding = 0u;
    Motor_SetDirection((target_position > current_position) ? 1 : 0);
    Motor_SetPWM(128u); /* Example PWM, TBD: tune per requirement */
}

uint8_t Control_Status(void) {
    if (movement_pending) {
        current_position = Feedback_Get();
        if (current_position == target_position) {
            holding = 1u;
            movement_pending = 0u;
            Motor_Disable();
        }
    }
    return holding ? 1u : 0u;
}

###### FILE: code/src/config_manager.c ######
#include "config_manager.h"
#include <stdint.h>
/* Loaded from Flash at startup [see datasheet Section 10, Table 38]: positions and mapping arrays are constant after init */
static const uint8_t command_map[6] = {0,1,2,3,4,5};
static const uint16_t position_thresholds[6] = {100, 400, 800, 1200, 1800, 2500}; /* Example values, [TBD—load from Flash per application] */

uint8_t Config_IsValidCommand(uint8_t cmd) {
    /* map: valid if entry exists in command_map for 0..5 */
    return (cmd < 6u);
}

uint8_t Config_MapFeedbackToPosition(uint16_t adc_sample) {
    uint8_t i;
    for (i = 0; i < 6u; ++i) {
        if (adc_sample < position_thresholds[i]) {
            return i;
        }
    }
    return 5u;
}
