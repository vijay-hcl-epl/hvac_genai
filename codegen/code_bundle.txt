###### FILE: code/src/flap_control_logic.c ######
#include "flap_control_logic.h"
#include "adc_interface.h"
#include "motor_driver_if.h"
#include "error_handler.h"
static uint8_t pos_target, pos_actual;
static FlapControlState state = FLAP_IDLE;
void FlapMoveTo(uint8_t pos) {
    pos_target = pos;
    pos_actual = ADCGetPos();
    if (pos_target != pos_actual && state != FLAP_SAFE && state != FLAP_ERROR) {
        state = FLAP_MOVING;
        if (pos_target > pos_actual)
            MotorDrive(MOTOR_FWD, 80);
        else
            MotorDrive(MOTOR_REV, 80);
    } else {
        state = FLAP_IDLE;
        MotorOff();
    }
}
FlapControlState FlapGetStatus(void) {
    return state;
}

static uint8_t error_flags;

void SetError(uint8_t code) {
    error_flags |= (1<<code);
    last_error_code = code;
    SetStatus(LED_ERR);
}

    error_flags = 0;
    SetStatus(LED_OK);
