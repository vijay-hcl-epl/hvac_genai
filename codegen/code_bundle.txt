#define UART_RX_LEN 32
#define CMD_HISTORY_LEN 8

static uint8_t cmd_buffer[UART_RX_LEN];
static uint8_t valid_cmd_flag = 0;
static uint8_t last_cmd_status = 0;
static uint8_t command_history[CMD_HISTORY_LEN];

void cmd_handler_init(void) {
    valid_cmd_flag = 0;
    last_cmd_status = 0;
}

void handle_uart_rx(uint8_t* buf, size_t len) {
    // Pseudocode: parse, check valid, update, pass to position control
    if(len == 0 || buf == NULL) {
        last_cmd_status = 1; // Error code
        send_error_status();
        return;
    }
    // Simple decode (stub)
    if(buf[0] > 0 && buf[0] <= 100) { // expected: command range is within 1..100 (example)
        valid_cmd_flag = 1;
        last_cmd_status = 0;
        command_history[0] = buf[0];
        set_target_position((int)buf[0]);
        send_status_response();
    } else {
        last_cmd_status = 1; // Invalid command
        send_error_status();
    }
}

uint8_t get_last_cmd_status(void) {
    return last_cmd_status;
}

void send_status_response(void) {
    // Send status over UART (stub)
    hal_uart_write(&last_cmd_status, 1);
}

void send_error_status(void) {
    uint8_t error = 1;
    hal_uart_write(&error, 1);
}

static int current_position = 0;
static uint8_t pos_verify_result = 0;

void position_control_init(void) {
    target_position = 0;
    current_position = 0;
    pos_verify_result = 0;
}

void set_target_position(int pos) {
    // Assume bounds 0..100 for example, real values: [TBD — Not found in technical source]
    if(pos < 0 || pos > 100) {
        pos_verify_result = 1; // Error
        return;
    }
    target_position = pos;
    start_drive((target_position > current_position) ? DIR_FORWARD : DIR_REVERSE);
}

uint8_t get_position_status(void) {
    return pos_verify_result;
}

void update_on_feedback(int pos) {
    current_position = pos;
    if(current_position == target_position) {
        stop_drive();
    }
}

        case LED_NORMAL: hal_gpio_set(10); break;

}

    if(key>=0 && key<PARAM_COUNT) param_store[key]=value;
}

int get_data(int key) {
    if(key>=0 && key<PARAM_COUNT) return param_store[key];
    return 0;
}
