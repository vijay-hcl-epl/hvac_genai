/**
 * @file command_parser.h
 * @brief Command Parser – UART byte parsing and validation for flap position commands.
 *
 * Software Unit: Command Parser
 * Traces: SWE-REQ-001, SWE-REQ-002, SWE-REQ-003, SWE-REQ-015, SWE-REQ-022
 *
 * Platform: STM32F407G-DISC1 / STM32 HAL
 *
 * ASSUMPTIONS (not verified on hardware):
 *   - UART handle (huart2) is initialised by system_init before use.
 *   - Position command byte values 0x01..0x05 map to FLAP_POS_1..FLAP_POS_5.
 *     Actual byte encoding is a design assumption; adjust to match the PC tool.
 */
#ifndef COMMAND_PARSER_H
#define COMMAND_PARSER_H

#include <stdint.h>
#include <stdbool.h>

/** Number of supported flap positions (design assumption: 5 discrete positions). */
#define CMD_NUM_POSITIONS  5U

/** Allowed single-byte position command values (1-based). */
#define CMD_POS_MIN        1U
#define CMD_POS_MAX        CMD_NUM_POSITIONS

/**
 * @brief Initialise the Command Parser internal state.
 *
 * Clears latest command and validity flag.
 * Called once from system_init().
 * Traces: SWE-REQ-020 (init)

/**
 * @brief Process one incoming UART byte.

 * Checks byte against the allowed position table.

 *
 * Traces: SWE-REQ-001, SWE-REQ-002, SWE-REQ-003, SWE-REQ-022
 *
 * @param rx_byte  The byte received from UART.
 */
void command_parser_process_byte(uint8_t rx_byte);

 * @brief Retrieve the latest validated position command.
 *
 * After reading, the valid flag is cleared (consumed).
 *
 * Traces: SWE-REQ-001, SWE-REQ-003, SWE-REQ-015
 *
 * @param[out] position  Pointer to store the commanded position (1..CMD_NUM_POSITIONS).
 * @return true if a new valid command was available; false otherwise.

/**
 * @file command_parser.c
 * @brief Command Parser implementation.
 *
 * Software Unit: Command Parser
 * Traces: SWE-REQ-001, SWE-REQ-002, SWE-REQ-003, SWE-REQ-015, SWE-REQ-022
 *
 * Platform: STM32F407G-DISC1 / STM32 HAL
 * No dynamic allocation (SWE-REQ-036).  All state is static.
 */

#include "command_parser.h"

/* ---- Static (module-level) data  [SWE-REQ-009, SWE-REQ-036, SWE-REQ-038] ---- */

/** Latest validated position command (1..CMD_NUM_POSITIONS). */
static uint8_t s_latest_position = 0U;

/** True when a new, unconsumed valid command is available. */
static bool    s_valid_flag      = false;

/* ---- Allowed position table (static mapping) [SWE-REQ-027, SWE-REQ-030] ---- */

/**
 * Lookup table of allowed command byte values.
 * Design assumption: bytes 0x01..0x05 represent positions 1..5.
 * Table stored in flash (const) – adjust if encoding changes.
 */
static const uint8_t s_allowed_positions[CMD_NUM_POSITIONS] =
{
    1U, 2U, 3U, 4U, 5U
};

/* ---- Private helpers ---- */

/**
 * @brief Check whether a byte is an allowed position command.
 * @return The position (1-based) or 0 if invalid.
 */
static uint8_t validate_byte(uint8_t byte)
{
    uint8_t i;
    for (i = 0U; i < CMD_NUM_POSITIONS; i++)
    {
        if (byte == s_allowed_positions[i])
        {
            return byte;  /* valid – value equals position number */
        }
    }
    return 0U;  /* invalid */
}

/* ---- Public API ---- */

void command_parser_init(void)
{
    s_latest_position = 0U;
    s_valid_flag      = false;
}

/*
 * SWE-REQ-001 : process single-byte UART command
 * SWE-REQ-002 : ignore invalid commands
 * SWE-REQ-003 : latest command wins (overwrites)
 * SWE-REQ-022 : constantly monitor UART input
 */
void command_parser_process_byte(uint8_t rx_byte)
{
    uint8_t pos = validate_byte(rx_byte);

    if (pos == 0U)
    {
        /* Invalid / unsupported – silently ignore [SWE-REQ-002] */
        return;
    }

    /* Store latest valid command (latest wins) [SWE-REQ-003] */
    s_latest_position = pos;
    s_valid_flag      = true;
}

bool command_parser_get_latest(uint8_t *position)
{
    if ((position == ((void *)0)) || (!s_valid_flag))
    {
        return false;
    }

    *position    = s_latest_position;
    s_valid_flag = false;  /* consumed */
    return true;
}

/**
 * @file feedback_processor.c
 * @brief Feedback Processor implementation.
 *
 * Software Unit: Feedback Processor
 * Traces: SWE-REQ-010, SWE-REQ-011, SWE-REQ-017, SWE-REQ-027, SWE-REQ-028, SWE-REQ-030
 *
 * Platform: STM32F407G-DISC1 / STM32 HAL (ADC1)
 * No dynamic allocation (SWE-REQ-036).  All state is static.
 *
 * ASSUMPTIONS / CALIBRATION NOTE:
 *   - ADC1 Channel 1 (PA1) is assumed for the potentiometer.
 *   - The 12-bit ADC range (0..4095) is divided into 5 equal zones.
 *     Threshold boundaries are PLACEHOLDERS and MUST be calibrated on
 *     the actual mechanical assembly before deployment.
 *   - hadc1 is declared extern; it is initialised in system_init.c.
 */

#include "feedback_processor.h"
#include "stm32f4xx_hal.h"

/* ---- Extern HAL handle (initialised in system_init.c) ---- */
extern ADC_HandleTypeDef hadc1;

/* ---- Static position boundary table [SWE-REQ-027, SWE-REQ-030] ---- */

/**
 * Upper ADC thresholds for each position zone (0-indexed).
 * Position 1 : ADC   0 ..  819
 * Position 2 : ADC 820 .. 1638
 * Position 3 : ADC 1639 .. 2457
 * Position 4 : ADC 2458 .. 3276
 * Position 5 : ADC 3277 .. 4095
 *
 * WARNING: These values are evenly spaced design placeholders.
 *          Calibrate for the actual potentiometer / gear linkage.
 */
static const uint16_t s_pos_upper_threshold[FB_NUM_POSITIONS] =
{
     819U,   /* Position 1 upper bound */
    1638U,   /* Position 2 upper bound */
    2457U,   /* Position 3 upper bound */
    3276U,   /* Position 4 upper bound */
    4095U    /* Position 5 upper bound */
};

/* ---- Module static state [SWE-REQ-009, SWE-REQ-036, SWE-REQ-038] ---- */

static uint16_t   s_raw_adc   = 0U;
static uint8_t    s_position  = 0U;
static bool       s_valid     = false;
static fb_state_t s_state     = FB_STATE_INIT;

/* ---- Private helpers ---- */

/**
 * @brief Map a validated ADC value to a logical position (1..FB_NUM_POSITIONS).
 * @param adc_val  Bounded ADC value (0..4095).
 * @return Position 1..5, or 0 on mapping error.
 */
static uint8_t map_adc_to_position(uint16_t adc_val)
{
    uint8_t i;
    for (i = 0U; i < FB_NUM_POSITIONS; i++)
    {
        if (adc_val <= s_pos_upper_threshold[i])
        {
            return (i + 1U);
        }
    }
    return 0U;  /* should not reach if thresholds cover full range */
}

/* ---- Public API ---- */

void feedback_processor_init(void)
{
    s_raw_adc  = 0U;
    s_position = 0U;
    s_valid    = false;
    s_state    = FB_STATE_INIT;
}

/*
 * SWE-REQ-010 : acquire ADC, interpret to logical position
 * SWE-REQ-011 : bound / validate ADC before use
 * SWE-REQ-017 : sample via ADC channel
 * SWE-REQ-027 : fixed static mapping
 * SWE-REQ-028 : no self-learning
 */
void feedback_processor_update(void)
{
    HAL_StatusTypeDef hal_status;
    uint32_t adc_raw_32;

    /* Start a single-conversion poll [SWE-REQ-017] */
    hal_status = HAL_ADC_Start(&hadc1);
    if (hal_status != HAL_OK)
    {
        s_valid = false;
        s_state = FB_STATE_INVALID;
        return;
    }

    hal_status = HAL_ADC_PollForConversion(&hadc1, 10U);  /* 10 ms timeout */
    if (hal_status != HAL_OK)
    {
        s_valid = false;
        s_state = FB_STATE_INVALID;
        HAL_ADC_Stop(&hadc1);
        return;
    }

    adc_raw_32 = HAL_ADC_GetValue(&hadc1);
    HAL_ADC_Stop(&hadc1);

    /* Bounding / validation [SWE-REQ-011] */
    if (adc_raw_32 > (uint32_t)FB_ADC_RAW_MAX)
    {
        s_valid = false;
        s_state = FB_STATE_INVALID;
        return;
    }

    s_raw_adc = (uint16_t)adc_raw_32;

    /* Map to position [SWE-REQ-010, SWE-REQ-027] */
    uint8_t pos = map_adc_to_position(s_raw_adc);
    if ((pos == 0U) || (pos > FB_NUM_POSITIONS))
    {
        s_valid = false;
        s_state = FB_STATE_INVALID;
        return;
    }

    s_position = pos;
    s_valid    = true;
    s_state    = FB_STATE_READY;
}

void feedback_processor_get_position(uint8_t *position, bool *valid)
{
    if (position != ((void *)0))
    {
        *position = s_position;
    }
    if (valid != ((void *)0))
    {
        *valid = s_valid;
    }
}

fb_state_t feedback_processor_get_state(void)
{
    return s_state;
}

/**
 * @file flap_control_logic.c
 * @brief Flap Control Logic implementation.
 *
 * Software Unit: Flap Control Logic
 * Traces: SWE-REQ-004, SWE-REQ-005, SWE-REQ-006, SWE-REQ-023, SWE-REQ-024, SWE-REQ-031
 *
 * Platform: STM32F407G-DISC1 (logic layer – no direct HW access)
 * No dynamic allocation (SWE-REQ-036).  All state is static.
 */

#include "flap_control_logic.h"
#include "feedback_processor.h"
#include "motor_driver.h"
#include "led_status_handler.h"

#include <stdbool.h>

/* ---- Module static state [SWE-REQ-009, SWE-REQ-036, SWE-REQ-038] ---- */

static flap_state_t s_state          = FLAP_STATE_IDLE;
static uint8_t      s_target_pos     = 0U;   /* 0 = no target yet */
static uint8_t      s_current_pos    = 0U;
static bool         s_has_target     = false;

/* ---- Public API ---- */

void flap_control_init(void)
{
    s_state       = FLAP_STATE_IDLE;
    s_target_pos  = 0U;
    s_current_pos = 0U;
    s_has_target  = false;
}

/*
 * SWE-REQ-004 : drive motor toward commanded position
 * SWE-REQ-006 : ignore if target == current
 * SWE-REQ-023 : compare command with feedback, move if needed
 * SWE-REQ-033 : initiate movement as soon as requirements satisfied
 */
void flap_control_issue_command(uint8_t target_pos)
{
    uint8_t fb_pos   = 0U;
    bool    fb_valid = false;

    feedback_processor_update();
    feedback_processor_get_position(&fb_pos, &fb_valid);

    if (!fb_valid)
    {
        /* Feedback fault – enter FAULT, stop motor [SWE-REQ-025, SWE-REQ-031] */
        motor_driver_stop();
        led_status_indicate_error();
        s_state = FLAP_STATE_FAULT;
        return;
    }

    s_current_pos = fb_pos;

    /* If target equals current position, ignore [SWE-REQ-006] */
    if (target_pos == s_current_pos)
    {
        motor_driver_stop();
        s_state = FLAP_STATE_TARGET_REACHED;
        led_status_set_position(s_current_pos);
        return;
    }

    /* Determine direction */
    s_target_pos = target_pos;
    s_has_target = true;

    if (s_target_pos > s_current_pos)
    {
        motor_driver_drive(MOTOR_DIR_FORWARD, 1U);
    }
    else
    {
        motor_driver_drive(MOTOR_DIR_REVERSE, 1U);
    }

    s_state = FLAP_STATE_MOVING;
}

/*
 * SWE-REQ-005 : cease actuation once target confirmed via feedback
 * SWE-REQ-008 : stop on invalid feedback or target reached
 * SWE-REQ-023 : compare command with feedback
 * SWE-REQ-024 : confirm target reached by position equivalence
 * SWE-REQ-025 : cease motor on error
 * SWE-REQ-031 : identify and handle out-of-range feedback
 */
void flap_control_run(void)
{
    uint8_t fb_pos   = 0U;
    bool    fb_valid = false;

    if (s_state == FLAP_STATE_IDLE || s_state == FLAP_STATE_TARGET_REACHED)
    {
        /* Nothing to do – waiting for a new command */
        return;
    }

    if (s_state == FLAP_STATE_FAULT)
    {
        /* Attempt recovery: re-read feedback */
        feedback_processor_update();
        feedback_processor_get_position(&fb_pos, &fb_valid);
        if (fb_valid)
        {
            s_state       = FLAP_STATE_IDLE;
            s_current_pos = fb_pos;
            led_status_set_position(fb_pos);
        }
        return;
    }

    /* FLAP_STATE_MOVING */
    feedback_processor_update();
    feedback_processor_get_position(&fb_pos, &fb_valid);

    if (!fb_valid)
    {
        /* Feedback fault while moving – safe stop [SWE-REQ-025, SWE-REQ-032] */
        motor_driver_stop();
        led_status_indicate_error();
        s_state = FLAP_STATE_FAULT;
        return;
    }

    s_current_pos = fb_pos;

    if (s_current_pos == s_target_pos)
    {
        /* Target reached [SWE-REQ-005, SWE-REQ-024] */
        motor_driver_stop();
        s_state = FLAP_STATE_TARGET_REACHED;
        s_has_target = false;
        led_status_set_position(s_current_pos);
    }
    else
    {
        /* Still moving – re-evaluate direction in case of overshoot */
        if (s_target_pos > s_current_pos)
        {
            motor_driver_drive(MOTOR_DIR_FORWARD, 1U);
        }
        else
        {
            motor_driver_drive(MOTOR_DIR_REVERSE, 1U);
        }
    }
}

flap_state_t flap_control_get_state(void)
{
    return s_state;
}

} led_pin_t;

/** Power LED pin. */
static const led_pin_t s_power_led = { GPIOD, GPIO_PIN_12 };

/** Position LED pins (index 0 = Position 1, etc.). */
static const led_pin_t s_pos_leds[LED_NUM_POSITIONS] =
{
    { GPIOD, GPIO_PIN_13 },  /* Position 1 */
    { GPIOD, GPIO_PIN_14 },  /* Position 2 */
    { GPIOD, GPIO_PIN_15 },  /* Position 3 */
    { GPIOE, GPIO_PIN_0  },  /* Position 4 */
    { GPIOE, GPIO_PIN_1  }   /* Position 5 */
};

/* ---- Module static state ---- */

static uint8_t s_current_position_led = 0U;  /* 0 = none lit */

/* ---- Private helpers ---- */

/** Turn all position LEDs OFF. */
static void all_position_leds_off(void)
{
    uint8_t i;
    for (i = 0U; i < LED_NUM_POSITIONS; i++)
    {
        HAL_GPIO_WritePin(s_pos_leds[i].port, s_pos_leds[i].pin, GPIO_PIN_RESET);
    }
    s_current_position_led = 0U;
}

/* ---- Public API ---- */

void led_status_init(void)
{
    /* All LEDs off at init */
    HAL_GPIO_WritePin(s_power_led.port, s_power_led.pin, GPIO_PIN_RESET);
    all_position_leds_off();
}

/* SWE-REQ-012 : power / running LED on */
void led_status_set_power(void)
{
    HAL_GPIO_WritePin(s_power_led.port, s_power_led.pin, GPIO_PIN_SET);
}

/*
 * SWE-REQ-013 : one-hot position LED
 * SWE-REQ-014 : alter LED only on valid event
 * SWE-REQ-039 : indicate position
 */
void led_status_set_position(uint8_t position)
{
    if (position == s_current_position_led)
    {
        return;  /* No change – SWE-REQ-014 */
    }

    all_position_leds_off();

    if ((position >= 1U) && (position <= LED_NUM_POSITIONS))
    {
        HAL_GPIO_WritePin(s_pos_leds[position - 1U].port,
                          s_pos_leds[position - 1U].pin,
                          GPIO_PIN_SET);
        s_current_position_led = position;
    }
}

/* SWE-REQ-032 : indicate error state */
void led_status_indicate_error(void)
{
    /* On error: turn off all position LEDs, leave power LED as-is. */
    all_position_leds_off();
}

/**
 * @file main.c
 * @brief Application entry point – init + super-loop task dispatcher.
 *
 * Traces: SWE-REQ-020, SWE-REQ-021, SWE-REQ-022, SWE-REQ-023, SWE-REQ-033, SWE-REQ-034
 *
 * Platform: STM32F407G-DISC1 / STM32 HAL / Bare-metal (no RTOS)
 * Execution model: simple while(1) super-loop per SWE-REQ-026 / SWE-REQ-034.
 *
 * The super-loop:
 *   1. Checks for a new UART byte (non-blocking).
 *   2. If a valid command is available, issues it to Flap Control.
 *   3. Runs the Flap Control periodic task (feedback check, state machine).
 *
 * No dynamic allocation.  No RTOS constructs.
 */

#include "system_init.h"
#include "command_parser.h"
#include "flap_control_logic.h"
#include "stm32f4xx_hal.h"

/* ---- Extern HAL UART handle (initialised in system_init.c) ---- */
extern UART_HandleTypeDef huart2;

/* ---- Static UART receive buffer [SWE-REQ-038] ---- */
static uint8_t s_uart_rx_byte = 0U;

/**
 * @brief Application entry point.
 *
 * Performs full system initialisation then enters the main dispatcher loop.
 */
int main(void)
{
    HAL_StatusTypeDef uart_status;
    uint8_t cmd_position = 0U;

    /* ---- STEP 1: System initialisation [SWE-REQ-020, SWE-REQ-021] ---- */
    system_init();

    /* ---- STEP 2: Main super-loop [SWE-REQ-022, SWE-REQ-034] ---- */
    while (1)
    {
        /*
         * 2a. Poll UART for a new byte (non-blocking, 0 ms timeout).
         * SWE-REQ-015, SWE-REQ-022
         */
        uart_status = HAL_UART_Receive(&huart2, &s_uart_rx_byte, 1U, 0U);
        if (uart_status == HAL_OK)
        {
            /* Feed byte to Command Parser [SWE-REQ-001] */
            command_parser_process_byte(s_uart_rx_byte);
        }

        /*
         * 2b. If a new valid command is ready, issue it.
         * SWE-REQ-003, SWE-REQ-004, SWE-REQ-023, SWE-REQ-033
         */
        if (command_parser_get_latest(&cmd_position))
        {
            flap_control_issue_command(cmd_position);
        }

        /*
         * 2c. Run Flap Control periodic task (feedback, state transitions).
         * SWE-REQ-005, SWE-REQ-008, SWE-REQ-023, SWE-REQ-024, SWE-REQ-025
         */
        flap_control_run();
    }

    /* Should never reach here – no return value needed in bare-metal. */
    /* return 0; */  /* MISRA: unreachable code omitted intentionally */
}

/**
 * @file motor_driver.c
 * @brief Motor Driver implementation for L298N via STM32 HAL GPIO + PWM.
 *
 * Software Unit: Motor Driver
 * Traces: SWE-REQ-007, SWE-REQ-008, SWE-REQ-016, SWE-REQ-025, SWE-REQ-036, SWE-REQ-038
 *
 * Platform: STM32F407G-DISC1 / STM32 HAL / L298N Dual H-Bridge
 * No dynamic allocation (SWE-REQ-036).  All state is static.
 *
 * ASSUMPTIONS / WIRING NOTE (not verified on hardware):
 *   - L298N IN1 → PB0  (GPIO output, motor direction A)
 *   - L298N IN2 → PB1  (GPIO output, motor direction B)
 *   - L298N ENA → PA0  (TIM2 CH1 PWM output, motor enable / speed)
 *   - PWM period and duty cycle values are design placeholders.
 *     Calibrate PWM frequency and duty for the 12 V DC geared motor.
 *   - htim2 is declared extern; it is initialised in system_init.c.
 */

#include "motor_driver.h"
#include "stm32f4xx_hal.h"

/* ---- Extern HAL timer handle (initialised in system_init.c) ---- */
extern TIM_HandleTypeDef htim2;

/* ---- Pin definitions (placeholder – match to actual wiring) ---- */

#define MOTOR_IN1_PORT   GPIOB
#define MOTOR_IN1_PIN    GPIO_PIN_0

#define MOTOR_IN2_PORT   GPIOB
#define MOTOR_IN2_PIN    GPIO_PIN_1

/* PWM channel for ENA on TIM2_CH1 */
#define MOTOR_PWM_CHANNEL  TIM_CHANNEL_1

/**
 * Configurable PWM duty cycle (0..htim2.Init.Period).
 * Design assumption: 70 % duty at Period=999 → CCR = 699.
 * SWE-REQ-035: document configurable responsiveness parameter.
 */
#define MOTOR_PWM_DUTY_DEFAULT  699U

/* ---- Module static state [SWE-REQ-009, SWE-REQ-036, SWE-REQ-038] ---- */

static motor_status_t s_status = MOTOR_STATUS_STOPPED;

/* ---- Public API ---- */

void motor_driver_init(void)
{
    /* Direction pins LOW (coast stop) */
    HAL_GPIO_WritePin(MOTOR_IN1_PORT, MOTOR_IN1_PIN, GPIO_PIN_RESET);
    HAL_GPIO_WritePin(MOTOR_IN2_PORT, MOTOR_IN2_PIN, GPIO_PIN_RESET);

    /* Ensure PWM output is stopped */
    HAL_TIM_PWM_Stop(&htim2, MOTOR_PWM_CHANNEL);

    s_status = MOTOR_STATUS_STOPPED;
}

/*
 * SWE-REQ-007 : set GPIO + PWM for direction/enable
 * SWE-REQ-016 : control motor via GPIO/PWM exclusively
 */
void motor_driver_drive(motor_dir_t dir, uint8_t enable)
{
    if (enable == 0U)
    {
        motor_driver_stop();
        return;
    }

    switch (dir)
    {
        case MOTOR_DIR_FORWARD:
            HAL_GPIO_WritePin(MOTOR_IN1_PORT, MOTOR_IN1_PIN, GPIO_PIN_SET);
            HAL_GPIO_WritePin(MOTOR_IN2_PORT, MOTOR_IN2_PIN, GPIO_PIN_RESET);
            break;

        case MOTOR_DIR_REVERSE:
            HAL_GPIO_WritePin(MOTOR_IN1_PORT, MOTOR_IN1_PIN, GPIO_PIN_RESET);
            HAL_GPIO_WritePin(MOTOR_IN2_PORT, MOTOR_IN2_PIN, GPIO_PIN_SET);
            break;

        case MOTOR_DIR_STOP:
        default:
            motor_driver_stop();
            return;
    }

    /* Set duty cycle and start PWM */
    __HAL_TIM_SET_COMPARE(&htim2, MOTOR_PWM_CHANNEL, MOTOR_PWM_DUTY_DEFAULT);
    HAL_TIM_PWM_Start(&htim2, MOTOR_PWM_CHANNEL);

    s_status = MOTOR_STATUS_RUNNING;
}

/* SWE-REQ-008, SWE-REQ-025, SWE-REQ-032 : immediate safe stop */
void motor_driver_stop(void)
{
    HAL_TIM_PWM_Stop(&htim2, MOTOR_PWM_CHANNEL);
    HAL_GPIO_WritePin(MOTOR_IN1_PORT, MOTOR_IN1_PIN, GPIO_PIN_RESET);
    HAL_GPIO_WritePin(MOTOR_IN2_PORT, MOTOR_IN2_PIN, GPIO_PIN_RESET);

    s_status = MOTOR_STATUS_STOPPED;
}

motor_status_t motor_driver_get_status(void)
{
    return s_status;
}
