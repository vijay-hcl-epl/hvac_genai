###### FILE: code/src/uart_command_handler.c ######
#include "uart_command_handler.h"
#include "control_logic_state_manager.h"
#include <string.h>
#include <ctype.h>

#define RX_BUF_LEN 4

static char rx_buffer[RX_BUF_LEN];
static int rx_index = 0;
static bool cmd_ready = false;
static int last_cmd_value = -1;

void uart_rx_handler(void *param) {
    char c = *(char *)param;
    if (cmd_ready) return; // Ignore until cleared by logic
    if (isdigit((unsigned char)c) && rx_index < 2) {
        rx_buffer[rx_index++] = c;
    } else if (c == '\n' || c == '\r') {
        rx_buffer[rx_index] = '\0';
        int val = -1;
        if (sscanf(rx_buffer, "%d", &val) == 1 && val >= 0 && val <= 5) {
            last_cmd_value = val;
            cmd_ready = true;
        }
        rx_index = 0;
        memset(rx_buffer, 0, sizeof(rx_buffer));
    } else {
        rx_index = 0;
        memset(rx_buffer, 0, sizeof(rx_buffer));
    }
}

bool uart_get_last_command(int *cmd_out) {
    if (cmd_ready && cmd_out) {
        *cmd_out = last_cmd_value;
        return true;
    }
    return false;
}

void uart_clear_command(void) {
    cmd_ready = false;
    last_cmd_value = -1;
}

#define CONTROL_LOGIC_STATE_MANAGER_H

    SYS_STATE_MOVE,
    SYS_STATE_ERROR
} system_state_t;

###### FILE: code/src/control_logic_state_manager.c ######
#include "control_logic_state_manager.h"
#include "uart_command_handler.h"
#include "position_feedback_handler.h"
#include "flap_motor_actuation_unit.h"
#include "status_position_indicator_unit.h"

static system_state_t sys_state = SYS_STATE_IDLE;
static int target_position = -1;
static int current_position = -1;

void control_logic_init(void) {
    sys_state = SYS_STATE_IDLE;
    target_position = -1;
    current_position = -1;
    status_led_set(true);
    position_indicator_update(current_position);
    motor_stop();
}

void control_logic_poll(void) {
    // Poll for new UART command
    int cmd;
    if (uart_get_last_command(&cmd)) {
        if (cmd >= 0 && cmd <= 5) {
            target_position = cmd;
            uart_clear_command();
            sys_state = SYS_STATE_MOVE;
        }
    }
    // Poll feedback
    position_feedback_poll();
    current_position = get_flap_position();
    if (get_adc_fault()) {
        sys_state = SYS_STATE_ERROR;
        motor_handle_error();
        return;
    }
    if (sys_state == SYS_STATE_MOVE) {
        if (current_position == target_position && current_position != -1) {
            motor_stop();
            sys_state = SYS_STATE_IDLE;
        } else if (current_position != -1) {
            motor_move_to_position(target_position, current_position);
        }
    } else if (sys_state == SYS_STATE_IDLE) {
        motor_stop();
    }
    position_indicator_update(current_position);
}

system_state_t control_logic_get_state(void) {
    return sys_state;
}
