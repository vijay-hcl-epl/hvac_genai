#include "user_command_if.h"

#include "error_handler.h"

#define CMD_MIN 0
#define CMD_MAX 5

static uint8_t cmd_buffer[8];
static uint8_t cmd_value = 0;
static uint8_t status_code = 0;
static uint8_t error_count = 0;

void UserCommandIF_Init(void) {
    cmd_value = 0;
    status_code = 0;
    error_count = 0;
}

void UserCommandIF_RxHandler(uint8_t rx_byte) {
    // For demo, take rx_byte as command directly
    if (rx_byte >= CMD_MIN && rx_byte <= CMD_MAX) {
        cmd_value = rx_byte;
        status_code = 0; // OK
        FlapControlLogic_RequestMoveTo(cmd_value);
    } else {
        status_code = 1; // ERR
        error_count++;
        ErrorHandler_SetErrorCode(ERR_CMD_INVALID);
    }
}

void UserCommandIF_Tick(void) {
    // No polling in this mock
}

void UserCommandIF_Reset(void) {
    cmd_value = 0;
    status_code = 0;
    error_count = 0;
}

uint8_t UserCommandIF_GetLastStatus(void) {
    return status_code;
}

uint8_t UserCommandIF_GetLastCommand(void) {
    return cmd_value;
}

#include "flap_control_logic.h"
#include "adc_if.h"
#include "motor_driver_if.h"
#include "error_handler.h"
#include "configuration_data.h"
#include <stdbool.h>

static uint8_t pos_target = 0;
static uint8_t pos_actual = 0;
static FlapControlState_t flap_state = FLAP_STATE_IDLE;
extern void StatusOutputIF_SetStatus(uint8_t code);

void FlapControlLogic_Init(void) {
    pos_target = 0;
    pos_actual = ADCIF_GetPos();
    flap_state = FLAP_STATE_IDLE;
}

void FlapControlLogic_RequestMoveTo(uint8_t position) {
    if (!ConfigurationData_IsValidPosition(position)) {
        flap_state = FLAP_STATE_ERROR;
        ErrorHandler_SetErrorCode(ERR_OUT_OF_RANGE);
        StatusOutputIF_SetStatus(STATUS_ERR);
        return;
    }
    pos_target = position;
    flap_state = FLAP_STATE_MOVING;
}

void FlapControlLogic_MainLoop(void) {
    pos_actual = ADCIF_GetPos();
    if (flap_state == FLAP_STATE_MOVING) {
        if (!ADCIF_IsValid()) {
            flap_state = FLAP_STATE_ERROR;
            ErrorHandler_SetErrorCode(ERR_ADC_FAULT);
            StatusOutputIF_SetStatus(STATUS_ERR);
            MotorDriverIF_Off();
        } else if (pos_actual < pos_target) {
            MotorDriverIF_Drive(DIRECTION_UP, 100);
            StatusOutputIF_SetStatus(STATUS_MOVING);
        } else if (pos_actual > pos_target) {
            MotorDriverIF_Drive(DIRECTION_DOWN, 100);
            StatusOutputIF_SetStatus(STATUS_MOVING);
        } else {
            MotorDriverIF_Off();
            flap_state = FLAP_STATE_IDLE;
            StatusOutputIF_SetStatus(STATUS_TARGET_REACHED);
        }
    } else if (flap_state == FLAP_STATE_IDLE) {
        MotorDriverIF_Off();
        StatusOutputIF_SetStatus(STATUS_IDLE);
    }
}

uint8_t FlapControlLogic_GetStatus(void) {
    return flap_state;
}

uint8_t FlapControlLogic_GetActualPosition(void) {
    return pos_actual;
}

###### FILE: code/src/configuration_data.c ######
#include "configuration_data.h"

static uint16_t thresholds[POS_NUM+1] = {0, 200, 400, 600, 800, 1000, 1200};

void ConfigurationData_Init(void) {}

bool ConfigurationData_IsValidPosition(uint8_t pos) {
    return (pos < POS_NUM);
}

bool ConfigurationData_IsInRange(uint16_t adc_val) {
    return (adc_val >= thresholds[0] && adc_val <= thresholds[POS_NUM]);
}

uint8_t ConfigurationData_MapAdcToPos(uint16_t adc_val) {
    for (uint8_t i = 0; i < POS_NUM; ++i) {
        if (adc_val < thresholds[i+1]) return i;
    }
    return POS_NUM - 1;
}
