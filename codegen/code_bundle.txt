#include "cmd_interp.h"
static uint8_t last_value = 0;
static uint8_t last_valid = 0;
cmd_status_t Cmd_decode(uint8_t data) {
    cmd_status_t ret;
    if (data <= 5) {
        last_value = data;
        last_valid = 1;
    } else {
        last_value = 0;
        last_valid = 0;
    }
    ret.value = last_value;
    ret.valid = last_valid;
    return ret;
}
cmd_status_t Cmd_status(void) {
    cmd_status_t ret = { last_value, last_valid };
    return ret;
}

flap_status_t FlapCtrl_eval(uint8_t cmd, uint16_t adc) {
    flap_status_t st;
    // [TBD — Not in source]: map ADC value to 0..5 pos
    // Check current vs. target and run/stop motor as needed
    if (cmd > 5) {
        flags = 1;
    } else {
        target = cmd;
        current = 0; // [TBD — Not in source]: ADC-to-pos(algo)
        if (current < target) Motor_run(MOTOR_CW, 50); // [TBD]: speed
        else if (current > target) Motor_run(MOTOR_CCW, 50);
        else Motor_stop();
        flags = 0;
    }
    st.target = target;
    st.current = current;
    st.flags = flags;
    return st;
}

    switch(event) {
        case 1: cur_state = S_MOVING; break;

#include "uart_lin_drv.h"
#include "adc_drv.h"
#include "motor_drv.h"
#include "gpio_hs_drv.h"
#include "cmd_interp.h"
#include "flap_ctrl.h"
#include "state_mgr.h"
#include "fault_handler.h"
#include "ui_handler.h"
#include "test_mon.h"

int main(void) {
    uart_lin_cfg_t uart_cfg = { UART_MODE, 19200, 0, 1 };
    Serial_init(&uart_cfg);
    adc_cfg_t adc_cfg = { 10, 0, 1 };
    ADC_init(&adc_cfg);
    motor_cfg_t motor_cfg = { 1000, 50, 0 };
    Motor_init(&motor_cfg);
    gpio_cfg_t gpio_cfg = {0, 1};
    GPIO_init(&gpio_cfg);
    UI_init();
    Test_enable();
    while (1) {
        uint8_t cmd = Serial_rx();
        cmd_status_t cstat = Cmd_decode(cmd);
        uint16_t adc_val = ADC_result();
        flap_status_t fstat = FlapCtrl_eval(cstat.value, adc_val);
        if (!cstat.valid) Fault_check(1);
        if (fstat.flags) StateMgr_update(3); // Fault
        else if (fstat.current == fstat.target) StateMgr_update(2); // Reached
        else StateMgr_update(1); // Moving
        UI_status(0);
        // Simple test monitor (not production-safe!)
        Test_query(0);
    }
    return 0;
}
